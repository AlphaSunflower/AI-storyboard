import copy

doc = Document("E:/Desktop/项目计划书.docx")

# 1) 锚点定位：按文本找段落（不要用固定索引）
def find_para(doc, text_prefix):
    for p in doc.paragraphs:
        if p.text.strip().startswith(text_prefix):
            return p
    raise AssertionError(f"锚点未找到: {text_prefix}")

# 2) 在锚点前顺序插入（addprevious 链保序，禁止 deepcopy+addnext）
def insert_before(anchor, template_p, text, style=None):
    new_p = copy.deepcopy(template_p._p)
    anchor._p.addprevious(new_p)
    from docx.text.paragraph import Paragraph
    np = Paragraph(new_p, anchor._parent)
    for r in list(np.runs[1:]):   # 保留首 run，清其余
        r._r.getparent().remove(r._r)
    if np.runs:
        np.runs[0].text = text
    else:
        np.add_run(text)
    if style:
        np.style = style
    return np

# 3) 标题段落：模板用同级别 Heading 段；正文用 Normal 段
# 4) 替换标题文本（重编号）：删 p.runs[1:]，改 p.runs[0].text（保留格式）
